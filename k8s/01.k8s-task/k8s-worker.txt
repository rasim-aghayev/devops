#!/bin/bash
set -euo pipefail

MASTER_IP="192.168.153.10"
KUBEADM_TOKEN="9ynbs4.g9bcv2xwvrhduj0w"
KUBEADM_DISCOVERY_HASH="d59ca20d205976951514d96cafbaed5305670a1d43917b543561f316b89bf653"  # только хэш без sha256:

echo "===== [1/9] System update and base packages ====="
apt update -y
apt install -y apt-transport-https ca-certificates curl gnupg

echo "===== [2/9] Adding Kubernetes repository ====="
mkdir -p /etc/apt/keyrings
curl -fsSL https://pkgs.k8s.io/core:/stable:/v1.30/deb/Release.key \
  | gpg --dearmor -o /etc/apt/keyrings/kubernetes-apt-keyring.gpg
chmod 644 /etc/apt/keyrings/kubernetes-apt-keyring.gpg

cat <<EOF >/etc/apt/sources.list.d/kubernetes.list
deb [signed-by=/etc/apt/keyrings/kubernetes-apt-keyring.gpg] https://pkgs.k8s.io/core:/stable:/v1.30/deb/ /
EOF

apt update -y

echo "===== [3/9] Installing kubelet, kubeadm, kubectl, containerd ====="
apt install -y kubelet kubeadm kubectl containerd
apt-mark hold kubelet kubeadm kubectl

echo "===== [4/9] Disabling swap ====="
swapoff -a || true
sed -i.bak '/swap/d' /etc/fstab || true

echo "===== [5/9] Enabling kernel modules ====="
cat <<EOF >/etc/modules-load.d/k8s.conf
overlay
br_netfilter
EOF
modprobe overlay
modprobe br_netfilter

echo "===== [6/9] Configuring sysctl ====="
cat <<EOF >/etc/sysctl.d/k8s.conf
net.ipv4.ip_forward=1
net.bridge.bridge-nf-call-iptables=1
net.bridge.bridge-nf-call-ip6tables=1
EOF
sysctl --system

echo "===== [7/9] Configuring containerd ====="
mkdir -p /etc/containerd
containerd config default > /etc/containerd/config.toml

# systemd cgroup driver
sed -i 's/SystemdCgroup = false/SystemdCgroup = true/' /etc/containerd/config.toml

# ensure SystemdCgroup inside runc.options
if ! grep -q 'SystemdCgroup = true' /etc/containerd/config.toml; then
    sed -i '/runc.options]/a \            SystemdCgroup = true' /etc/containerd/config.toml
fi

# required fix for Kubernetes 1.30 (pause container)
sed -i 's@sandbox_image = ".*"@sandbox_image = "registry.k8s.io/pause:3.9"@' /etc/containerd/config.toml

systemctl daemon-reload
systemctl restart containerd
systemctl enable containerd
systemctl enable kubelet

echo "===== [8/9] Checking whether node is already joined ====="
if [[ -f /etc/kubernetes/kubelet.conf ]]; then
  echo "Node already appears to be part of a cluster."
  echo "If rejoin needed:"
  echo "  kubeadm reset -f"
  echo "  rm -rf /etc/cni/net.d"
  exit 0
fi

echo "===== [9/9] Joining Kubernetes cluster ====="

if [[ "$KUBEADM_TOKEN" == "<TOKEN>" || "$KUBEADM_DISCOVERY_HASH" == "<SHA256_HASH>" ]]; then
  echo "❌ ERROR: Token or discovery hash is not set."
  echo "Run on master:"
  echo "    kubeadm token create --print-join-command"
  exit 1
fi

kubeadm join "${MASTER_IP}:6443" \
  --token "${KUBEADM_TOKEN}" \
  --discovery-token-ca-cert-hash sha256:"${KUBEADM_DISCOVERY_HASH}"

echo ""
echo "🎉 WORKER NODE SUCCESSFULLY JOINED!"
echo ""
